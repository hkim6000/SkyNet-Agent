        private const int Max = 12;

        private static string Thumb(List<MediaItem> media, string path)
        {
            MediaItem? x = media.FirstOrDefault(i => i.Path == path);
            return x != null && x.Thumb != string.Empty ? x.Thumb : path;
        }


        private static string StripItem(List<MediaItem> media, string path, bool canE)
        {
            return "<div class=\"xx-si\" data-src=\"" + Ui.Esc(path) + "\"><img src=\"" + Ui.Esc(Thumb(media, path)) + "\" alt=\"\"><i class=\"xx-sn\"></i>" + (canE ? "<div class=\"xx-sctl\"><button type=\"button\" aria-label=\"Move earlier\" onclick=\"AdminThingsJs.sMove(this,-1)\">&#9664;</button><button type=\"button\" aria-label=\"Move later\" onclick=\"AdminThingsJs.sMove(this,1)\">&#9654;</button><button type=\"button\" aria-label=\"Remove\" onclick=\"AdminThingsJs.sDrop(this)\">&#10005;</button></div>" : string.Empty) + "</div>";
        }

        public async Task<ApiResponse> Pick()
        {
            ApiResponse response = new ApiResponse();
            Ctx? c = await Guard(response);
            if (c == null)
            {
                return response;
            }
            if (!Access.Can(c.U, AreaKey, 'E'))
            {
                return await Refuse(response, c, "pick photos", string.Empty, "You can look but not change this.");
            }
            List<MediaItem> media = await Media.Load();
            response.SetElementContents("xx-pkgrid", PickGrid(media, Ui.Clip(GetDataValue("pterm"), 40)));
            response.ExecuteScript("AdminThingsJs.pkListed();");
            return response;
        }

        private static string PickGrid(List<MediaItem> media, string term)
        {
            List<MediaItem> rows = media.Where(m => m.Kind == "photo" && (term == string.Empty || Ui.Has(m.Name, term) || Ui.Has(m.Alt, term))).OrderBy(m => m.Name, StringComparer.Ordinal).Take(400).ToList();
            if (rows.Count == 0)
            {
                return "<p class=\"xx-hint\">Nothing matches. Upload a photo above, or add photos in the Media library.</p>";
            }
            return string.Concat(rows.Select(m => "<button type=\"button\" class=\"xx-pki\" data-src=\"" + Ui.Esc(m.Path) + "\" data-thumb=\"" + Ui.Esc(m.Thumb != string.Empty ? m.Thumb : m.Path) + "\" onclick=\"AdminThingsJs.pkToggle(this)\" title=\"" + Ui.Esc(m.Alt != string.Empty ? m.Alt : m.Name) + "\"><img src=\"" + Ui.Esc(m.Thumb != string.Empty ? m.Thumb : m.Path) + "\" alt=\"\" loading=\"lazy\"><i class=\"xx-pkn\"></i><span>" + Ui.Esc(m.Name) + "</span></button>"));
        }

        public async Task<ApiResponse> Upload()
        {
            ApiResponse response = new ApiResponse();
            Ctx? c = await Guard(response);
            if (c == null)
            {
                return response;
            }
            if (!Access.Can(c.U, AreaKey, 'E') || !Access.Can(c.U, "media", 'E'))
            {
                response.ExecuteScript("AdminThingsJs.pkUploaded('You cannot upload to the media library.');");
                return await Refuse(response, c, "upload media", string.Empty, "You cannot upload to the media library.");
            }
            SiteData site = await Store.LoadSite();
            List<Incoming> files = await Media.Received();
            if (files.Count == 0)
            {
                response.ExecuteScript("AdminThingsJs.pkUploaded('Choose a photo first.');");
                return response;
            }
            (int added, List<string> errors) = await Media.AddAll(files, site.MaxVideoMb, c.U.Name);
            if (added > 0)
            {
                await Store.Log(c.U.Id, "upload media", Ui.Count(added, "file", "files"), true);
            }
            List<MediaItem> media = await Media.Load();
            response.SetElementContents("xx-pkgrid", PickGrid(media, string.Empty));
            response.ExecuteScript("AdminThingsJs.pkUploaded('" + Ui.Js(((added > 0 ? Ui.Count(added, "file", "files") + " added to the library. " : string.Empty) + string.Join(" ", errors)).Trim()) + "');");
            return response;
        }

    function esc(t) {
        return String(t == null ? '' : t).replace(/[&<>"']/g, function (ch) {
            return { '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;' }[ch];
        });
    }

    function touch() {
        markDirty();
    }

   Send the list with the form: list.push({ key: 'photos', vlu: JSON.stringify(stripPaths()) });

    function stripPaths() {
        return [].map.call(document.querySelectorAll('#xx-strip .xx-si'), function (s) {
            return s.getAttribute('data-src');
        });
    }

    function numberStrip() {
        var items = document.querySelectorAll('#xx-strip .xx-si');
        [].forEach.call(items, function (s, i) {
            s.querySelector('.xx-sn').textContent = i + 1;
        });
        var c = el('xx-pcount');
        if (c) {
            c.textContent = items.length;
        }
    }

    function sMove(btn, d) {
        var s = btn.closest('.xx-si');
        if (d < 0 && s.previousElementSibling) {
            s.parentNode.insertBefore(s, s.previousElementSibling);
        } else if (d > 0 && s.nextElementSibling) {
            s.parentNode.insertBefore(s.nextElementSibling, s);
        }
        numberStrip();
        touch();
    }

    function sDrop(btn) {
        btn.closest('.xx-si').remove();
        numberStrip();
        touch();
    }

    function stripItem(src, thumb) {
        return '<div class="xx-si" data-src="' + esc(src) + '"><img src="' + esc(thumb) + '" alt=""><i class="xx-sn"></i><div class="xx-sctl"><button type="button" aria-label="Move earlier" onclick="AdminThingsJs.sMove(this,-1)">&#9664;</button><button type="button" aria-label="Move later" onclick="AdminThingsJs.sMove(this,1)">&#9654;</button><button type="button" aria-label="Remove" onclick="AdminThingsJs.sDrop(this)">&#10005;</button></div></div>';
    }

    function pickPhotos() {
        pkSel = stripPaths();
        el('xx-pkq').value = '';
        el('xx-pkmsg').textContent = '';
        var d = el('xx-pk');
        d.classList.add('xx-open');
        d.setAttribute('aria-hidden', 'false');
        pkList();
    }

    function pkList() {
        api('Pick', [{ key: 'pterm', vlu: (el('xx-pkq').value || '').trim() }]);
    }

    function pkSearch() {
        clearTimeout(pkTimer);
        pkTimer = setTimeout(pkList, 300);
    }

    function pkMark() {
        [].forEach.call(document.querySelectorAll('#xx-pkgrid .xx-pki'), function (b) {
            var i = pkSel.indexOf(b.getAttribute('data-src'));
            b.classList.toggle('xx-on', i >= 0);
            b.setAttribute('aria-pressed', i >= 0 ? 'true' : 'false');
            b.querySelector('.xx-pkn').textContent = i >= 0 ? i + 1 : '';
        });
        el('xx-pkcount').textContent = pkSel.length + ' of ' + MAX + ' chosen. Click a photo to add or remove it; the numbers show the order.';
    }

    function pkListed() {
        pkMark();
    }

    function pkToggle(b) {
        var src = b.getAttribute('data-src'), i = pkSel.indexOf(src);
        if (i >= 0) {
            pkSel.splice(i, 1);
        } else if (pkSel.length >= MAX) {
            toast('Up to ' + MAX + ' photos. Remove one first.');
            return;
        } else {
            pkSel.push(src);
        }
        pkMark();
    }

    function pkUse() {
        var thumbs = {};
        [].forEach.call(document.querySelectorAll('#xx-strip .xx-si'), function (s) {
            thumbs[s.getAttribute('data-src')] = s.querySelector('img').getAttribute('src');
        });
        [].forEach.call(document.querySelectorAll('#xx-pkgrid .xx-pki'), function (b) {
            thumbs[b.getAttribute('data-src')] = b.getAttribute('data-thumb');
        });
        el('xx-strip').innerHTML = pkSel.map(function (s) {
            return stripItem(s, thumbs[s] || s);
        }).join('');
        numberStrip();
        touch();
        pkClose();
    }

    function pkOpen() {
        var d = el('xx-pk');
        return !!d && d.classList.contains('xx-open');
    }

    function pkClose() {
        var d = el('xx-pk');
        if (d) {
            d.classList.remove('xx-open');
            d.setAttribute('aria-hidden', 'true');
        }
        el('xx-pkfile').value = '';
    }

    function pkUpload() {
        el('xx-pkmsg').textContent = 'Uploading.';
        api('Upload');
    }

    function pkUploaded(msg) {
        el('xx-pkfile').value = '';
        el('xx-pkmsg').textContent = msg;
        pkMark();
    }
